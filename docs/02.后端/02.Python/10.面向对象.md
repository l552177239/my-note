### 1. 封装（Encapsulation）

将数据和操作数据的方法绑定在一起，隐藏内部实现细节，只暴露必要的接口。

```python
class BankAccount:
    """银行账户类 - 封装示例"""
    
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.__balance = balance  # 私有属性
        self.__transaction_history = []  # 私有属性
    
    def deposit(self, amount):
        """存款"""
        if amount <= 0:
            raise ValueError("存款金额必须大于0")
        self.__balance += amount
        self.__transaction_history.append(f"+{amount}")
        return self.__balance
    
    def withdraw(self, amount):
        """取款"""
        if amount <= 0:
            raise ValueError("取款金额必须大于0")
        if amount > self.__balance:
            raise ValueError("余额不足")
        self.__balance -= amount
        self.__transaction_history.append(f"-{amount}")
        return self.__balance
    
    @property
    def balance(self):
        """只读属性：余额"""
        return self.__balance
    
    @property
    def history(self):
        """只读属性：交易记录"""
        return self.__transaction_history.copy()

# 使用
account = BankAccount("Alice", 1000)
account.deposit(500)
account.withdraw(200)
print(account.balance)      # 1300
print(account.history)      # ['+500', '-200']
# account.__balance = 9999  # 无法直接修改
```

### 2. 继承（Inheritance）

子类继承父类的属性和方法，可以扩展或重写。

```python
# 父类（基类）
class Animal:
    def __init__(self, name):
        self.name = name
    
    def speak(self):
        raise NotImplementedError("子类必须实现此方法")
    
    def introduce(self):
        return f"我是 {self.name}"

# 子类（派生类）
class Dog(Animal):
    def __init__(self, name, breed):
        super().__init__(name)  # 调用父类构造方法
        self.breed = breed
    
    def speak(self):
        return f"{self.name} 说：汪汪！"
    
    def fetch(self):
        return f"{self.name} 去捡球了"

class Cat(Animal):
    def speak(self):
        return f"{self.name} 说：喵喵~"

# 使用
dog = Dog("Buddy", "Golden Retriever")
cat = Cat("Kitty")

print(dog.speak())      # Buddy 说：汪汪！
print(dog.introduce())  # 我是 Buddy
print(dog.fetch())      # Buddy 去捡球了
print(cat.speak())      # Kitty 说：喵喵~
```

#### 多继承

```python
class Flyable:
    def fly(self):
        return "我能飞！"

class Swimmable:
    def swim(self):
        return "我能游泳！"

class Duck(Animal, Flyable, Swimmable):
    def speak(self):
        return f"{self.name} 说：嘎嘎！"

duck = Duck("Donald")
print(duck.fly())       # 我能飞！
print(duck.swim())      # 我能游泳！

print(Duck.__mro__)
```

### 3. 多态（Polymorphism）

不同的对象对同一消息（方法调用）做出不同的响应。Python 中通过鸭子类型实现。

```python
# 多态：不同对象调用同一个方法，产生不同结果
class Dog(Animal):
    def speak(self):
        return "汪汪！"

class Cat(Animal):
    def speak(self):
        return "喵喵~"

class Duck(Animal):
    def speak(self):
        return "嘎嘎！"

# 多态函数：不关心具体类型，只关心有没有 speak 方法
def animal_concert(animals):
    for animal in animals:
        print(animal.speak())

# 使用
zoo = [Dog("Buddy"), Cat("Kitty"), Duck("Donald")]
animal_concert(zoo)
# 汪汪！
# 喵喵~
# 嘎嘎！

# 鸭子类型：不检查类型，只检查行为
def make_it_speak(obj):
    """只要对象有 speak 方法就行"""
    return obj.speak()

class Robot:
    def speak(self):
        return "哔哔啵啵"

print(make_it_speak(Robot()))  # 哔哔啵啵
```

### 4. 抽象基类（ABC）

强制子类实现特定方法。

```python
from abc import ABC, abstractmethod

class Shape(ABC):
    """抽象基类"""
    
    @abstractmethod
    def area(self):
        """子类必须实现面积计算"""
        pass
    
    @abstractmethod
    def perimeter(self):
        """子类必须实现周长计算"""
        pass

class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height
    
    def area(self):
        return self.width * self.height
    
    def perimeter(self):
        return 2 * (self.width + self.height)

class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius
    
    def area(self):
        import math
        return math.pi * self.radius ** 2
    
    def perimeter(self):
        import math
        return 2 * math.pi * self.radius

# 使用
rect = Rectangle(3, 4)
circle = Circle(5)
print(rect.area())    # 12
print(circle.area())  # 78.54...

# shape = Shape()  # 不能实例化抽象类
```

### 5. 综合示例：面向对象设计

```python
from abc import ABC, abstractmethod
from datetime import datetime

# 抽象基类
class Employee(ABC):
    def __init__(self, name, employee_id):
        self.name = name
        self.employee_id = employee_id
        self._join_date = datetime.now()
    
    @abstractmethod
    def calculate_salary(self):
        pass
    
    def get_info(self):
        return f"[{self.employee_id}] {self.name}"

# 全职员工
class FullTimeEmployee(Employee):
    def __init__(self, name, employee_id, monthly_salary):
        super().__init__(name, employee_id)
        self.monthly_salary = monthly_salary
    
    def calculate_salary(self):
        return self.monthly_salary

# 兼职员工
class PartTimeEmployee(Employee):
    def __init__(self, name, employee_id, hourly_rate, hours_worked):
        super().__init__(name, employee_id)
        self.hourly_rate = hourly_rate
        self.hours_worked = hours_worked
    
    def calculate_salary(self):
        return self.hourly_rate * self.hours_worked

# 销售员工
class SalesEmployee(Employee):
    def __init__(self, name, employee_id, base_salary, sales_amount, commission_rate):
        super().__init__(name, employee_id)
        self.base_salary = base_salary
        self.sales_amount = sales_amount
        self.commission_rate = commission_rate
    
    def calculate_salary(self):
        return self.base_salary + self.sales_amount * self.commission_rate

# 工资系统
class PayrollSystem:
    def __init__(self):
        self.employees = []
    
    def add_employee(self, employee):
        self.employees.append(employee)
    
    def calculate_total_payroll(self):
        total = 0
        for emp in self.employees:
            salary = emp.calculate_salary()  # 多态调用
            print(f"{emp.get_info()}: CNY{salary:,.2f}")
            total += salary
        return total

# 使用
payroll = PayrollSystem()
payroll.add_employee(FullTimeEmployee("Alice", "E001", 15000))
payroll.add_employee(PartTimeEmployee("Bob", "E002", 100, 80))
payroll.add_employee(SalesEmployee("Charlie", "E003", 8000, 50000, 0.05))

total = payroll.calculate_total_payroll()
print(f"\n总工资支出: CNY{total:,.2f}")
```

### 6. 面向对象速查表

| 特性 | 概念 | Python 实现 |
|------|------|-------------|
| **封装** | 隐藏内部细节，暴露接口 | 私有属性 `__`、 `@property` |
| **继承** | 代码复用，建立类层次 | `class Child(Parent)`、`super()` |
| **多态** | 同一接口，不同实现 | 鸭子类型、`@abstractmethod` |
| **抽象** | 定义接口规范 | `ABC` 抽象基类 |
| **组合** | 有一个关系 | 将一个类的实例作为另一个类的属性 |

> 组合优于继承：当类之间不是严格的 is-a 关系时，优先考虑组合而非继承。
