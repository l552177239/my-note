# 文件操作

<!-- more -->

### 1. 打开与关闭文件

Python 使用内置的 `open()` 函数打开文件，操作完成后需要关闭。

```python
# 传统方式（不推荐，容易忘记关闭）
f = open("data.txt", "r", encoding="utf-8")
content = f.read()
f.close()

# 推荐方式：with 语句（自动关闭，即使出错也会关闭）
with open("data.txt", "r", encoding="utf-8") as f:
    content = f.read()
# 离开 with 块后，文件自动关闭
```

### 2. 文件打开模式

| 模式 | 含义 | 说明 |
|------|------|------|
| `"r"` | 只读 | 默认模式，文件必须存在 |
| `"w"` | 写入 | 覆盖写入，文件不存在则创建 |
| `"a"` | 追加 | 在文件末尾添加内容 |
| `"x"` | 独占创建 | 文件已存在则报错 |
| `"b"` | 二进制 | 与其他模式组合，如 `"rb"`、`"wb"` |
| `"+"` | 读写 | 与其他模式组合，如 `"r+"`、`"w+"` |

```python
# 只读
with open("data.txt", "r", encoding="utf-8") as f:
    data = f.read()

# 覆盖写入（会清空原内容）
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("新内容\n")

# 追加（保留原内容，在末尾添加）
with open("log.txt", "a", encoding="utf-8") as f:
    f.write("日志记录\n")

# 读写模式
with open("data.txt", "r+", encoding="utf-8") as f:
    content = f.read()
    f.write("\n追加内容")

# 二进制模式（图片、视频等）
with open("image.png", "rb") as f:
    image_data = f.read()
```

### 3. 读取文件内容

```python
# 方法 1：读取全部内容
with open("data.txt", "r", encoding="utf-8") as f:
    content = f.read()
    print(content)

# 方法 2：逐行读取（推荐大文件使用）
with open("data.txt", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())  # strip() 去除首尾空白和换行

# 方法 3：读取为列表
with open("data.txt", "r", encoding="utf-8") as f:
    lines = f.readlines()  # 每行作为一个元素，包含 \n

# 方法 4：读取指定字节数
with open("data.txt", "r", encoding="utf-8") as f:
    chunk = f.read(1024)  # 读取 1024 个字符

# 方法 5：逐行读取（不加载到内存）
with open("big_file.txt", "r", encoding="utf-8") as f:
    while True:
        line = f.readline()
        if not line:
            break
        process(line)
```

### 4. 写入文件内容

```python
# 写入字符串
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("第一行\n")
    f.write("第二行\n")

# 写入多行
lines = ["第一行\n", "第二行\n", "第三行\n"]
with open("output.txt", "w", encoding="utf-8") as f:
    f.writelines(lines)

# 追加内容
with open("log.txt", "a", encoding="utf-8") as f:
    f.write("新日志\n")
```

### 5. 文件路径处理

```python
import os
from pathlib import Path

# 使用 pathlib（推荐，跨平台）
base_dir = Path(__file__).parent
data_file = base_dir / "data" / "users.txt"

# 判断存在性
print(data_file.exists())       # True / False
print(data_file.is_file())      # 是否是文件
print(data_file.is_dir())       # 是否是目录

# 创建目录
data_dir = base_dir / "output"
data_dir.mkdir(parents=True, exist_ok=True)

# 获取文件信息
print(data_file.name)           # users.txt
print(data_file.stem)           # users
print(data_file.suffix)         # .txt
print(data_file.parent)         # .../data

# 使用 os 模块
print(os.getcwd())              # 当前工作目录
print(os.listdir("."))          # 列出目录内容
os.makedirs("new_folder", exist_ok=True)
os.remove("file.txt")
os.rename("old.txt", "new.txt")
```

### 6. CSV 文件操作

```python
import csv

# 写入 CSV
with open("users.csv", "w", encoding="utf-8", newline="") as f:
    writer = csv.writer(f)
    writer.writerow(["姓名", "年龄", "城市"])
    writer.writerow(["张三", 25, "北京"])
    writer.writerow(["李四", 30, "上海"])

# 读取 CSV
with open("users.csv", "r", encoding="utf-8") as f:
    reader = csv.reader(f)
    for row in reader:
        print(row)

# 字典方式读写
with open("users.csv", "r", encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(f"{row['姓名']} - {row['年龄']}岁")

with open("users.csv", "w", encoding="utf-8", newline="") as f:
    fieldnames = ["姓名", "年龄", "城市"]
    writer = csv.DictWriter(f, fieldnames=fieldnames)
    writer.writeheader()
    writer.writerow({"姓名": "王五", "年龄": 28, "城市": "广州"})
```

### 7. JSON 文件操作

```python
import json

data = {
    "name": "Alice",
    "age": 20,
    "hobbies": ["reading", "coding"],
    "address": {"city": "Beijing", "zip": "100000"}
}

# 写入 JSON
with open("data.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=4)

# 读取 JSON
with open("data.json", "r", encoding="utf-8") as f:
    loaded_data = json.load(f)
    print(loaded_data["name"])

# 字符串与 JSON 互转
json_str = json.dumps(data, ensure_ascii=False)
obj = json.loads(json_str)
```

### 8. 文件操作综合示例

```python
from pathlib import Path
import json
import shutil

# 批量处理文本文件
def process_text_files(folder_path):
    folder = Path(folder_path)
    results = []
    for file_path in folder.glob("*.txt"):
        with open(file_path, "r", encoding="utf-8") as f:
            content = f.read()
            results.append({
                "file": file_path.name,
                "words": len(content.split()),
                "chars": len(content)
            })
    with open("stats.json", "w", encoding="utf-8") as f:
        json.dump(results, f, ensure_ascii=False, indent=2)
    return results

# 安全复制文件
def safe_copy(src, dst):
    src_path, dst_path = Path(src), Path(dst)
    if not src_path.exists():
        raise FileNotFoundError(f"源文件不存在: {src}")
    dst_path.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(src_path, dst_path)
```
