# Pythonic 编程风格

## 1. 代码风格指南

```python
# 命名规范
module_name          # 模块：小写+下划线
package_name         # 包：小写，简短
ClassName            # 类：大驼峰
method_name          # 方法：小写+下划线
function_name        # 函数：小写+下划线
GLOBAL_CONSTANT      # 常量：全大写
_private_var         # 私有变量：前导下划线
__private_var        # 名称改写：双前导下划线

# 缩进：4个空格（不要用Tab）
# 行长度：最大79字符（现代可放宽到100-120）
```

## 2. Pythonic 写法

```python
# 交换变量
a, b = b, a

# 多重赋值
x, y, z = 1, 2, 3

# 解包
first, *rest = [1, 2, 3, 4, 5]
head, *body, tail = [1, 2, 3, 4, 5]

# 合并字典（Python 3.9+）
merged = dict_a | dict_b

# 判断非空
if items: pass

# 获取字典值
count = d.get(key, 0)

# 列表推导式
result = [x * 2 for x in items if x > 0]

# 使用 enumerate
for i, item in enumerate(items, start=1):
    pass

# 使用 zip
for name, score in zip(names, scores):
    pass

# 使用 any/all
if any(x > 0 for x in numbers): pass
if all(x > 0 for x in numbers): pass

# EAFP 风格（请求原谅比请求许可更容易）
try:
    value = d[key]
except KeyError:
    value = default

# 链式比较
if 0 < x < 10: pass

# 使用 __main__ 保护
if __name__ == "__main__":
    main()
```

## 3. 常用惯用法

```python
# 列表去重保持顺序
unique = list(dict.fromkeys([1, 2, 2, 3, 3, 3]))

# 扁平化列表
flat = [item for sublist in nested for item in sublist]

# 计数器
from collections import Counter
counts = Counter(["a", "b", "a", "c", "a", "b"])
print(counts.most_common(2))

# 默认字典
from collections import defaultdict
d = defaultdict(list)
d["fruits"].append("apple")

# 双端队列
from collections import deque
dq = deque(maxlen=100)
dq.append(1)
dq.appendleft(2)

# 命名元组
from collections import namedtuple
Point = namedtuple("Point", ["x", "y"])
p = Point(3, 4)

# 数据类
from dataclasses import dataclass

@dataclass
class User:
    name: str
    age: int = 0
    email: str = ""

user = User("Alice", 25, "alice@example.com")
print(user)
```
