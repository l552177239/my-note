# 性能优化

## 1. 性能分析

```python
import cProfile
import timeit

def slow_function():
    result = []
    for i in range(10000):
        result.append(i ** 2)
    return result

cProfile.run("slow_function()")
print(timeit.timeit("[i**2 for i in range(1000)]", number=100))

# 内存分析
import tracemalloc
tracemalloc.start()
current, peak = tracemalloc.get_traced_memory()
print(f"当前内存: {current / 1024 / 1024:.2f} MB")
tracemalloc.stop()
```

## 2. 优化技巧

```python
# 1. 列表推导式替代循环
squares = [i ** 2 for i in range(1000)]

# 2. 生成器表达式处理大数据
sum_of_squares = sum(i ** 2 for i in range(1000000))

# 3. 使用 set/dict 进行查找（O(1)）
lookup = set([1, 2, 3, 4, 5])
if 3 in lookup: pass

# 4. 使用 lru_cache 缓存
from functools import lru_cache

@lru_cache(maxsize=128)
def fibonacci(n):
    if n < 2: return n
    return fibonacci(n - 1) + fibonacci(n - 2)

# 5. 使用 __slots__ 减少内存
class Point:
    __slots__ = ["x", "y"]
    def __init__(self, x, y):
        self.x = x
        self.y = y

# 6. 字符串拼接用 join
result = "".join([str(i) for i in range(100)])

# 7. 多进程处理 CPU 密集型任务
from multiprocessing import Pool
def process(data):
    return sum(x ** 2 for x in data)
```
