### 1. 定义函数

```python
# 基础函数
def greet(name):
    """向用户问好"""
    return f"Hello, {name}!"

# 调用
message = greet("Alice")
print(message)  # Hello, Alice!

# 无返回值函数
def print_info():
    print("这是一个信息")

# 多个返回值（实际是返回元组）
def get_min_max(nums):
    return min(nums), max(nums)

minimum, maximum = get_min_max([3, 1, 4, 1, 5])
```

### 2. 参数类型

```python
# 位置参数
def add(a, b):
    return a + b

add(3, 5)  # 8

# 默认参数
def greet(name, greeting="Hello"):
    return f"{greeting}, {name}!"

greet("Alice")           # Hello, Alice!
greet("Alice", "Hi")     # Hi, Alice!

# 默认参数不要用可变对象！
def bad_append(item, lst=[]):  # 错误！
    lst.append(item)
    return lst

def good_append(item, lst=None):  # 正确
    if lst is None:
        lst = []
    lst.append(item)
    return lst

# 关键字参数
def create_user(name, age, city="Beijing"):
    return {"name": name, "age": age, "city": city}

create_user(age=20, name="Alice")
create_user("Alice", 20, city="Shanghai")

# 可变位置参数 *args
def sum_all(*args):
    return sum(args)

sum_all(1, 2, 3, 4)  # 10
sum_all()            # 0

# 可变关键字参数 **kwargs
def print_info(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

print_info(name="Alice", age=20, city="Beijing")

# 组合使用
def func(a, b, *args, c=10, **kwargs):
    print(f"a={a}, b={b}, args={args}, c={c}, kwargs={kwargs}")

func(1, 2, 3, 4, c=20, d=5, e=6)
# a=1, b=2, args=(3, 4), c=20, kwargs={'d': 5, 'e': 6}

# 仅限位置参数和仅限关键字参数（Python 3.8+）
def func(pos_only, /, pos_or_kwd, *, kwd_only):
    pass

func(1, 2, kwd_only=3)       # 正确
func(1, pos_or_kwd=2, kwd_only=3)  # 正确
# func(pos_only=1, ...)      # 错误！
```

### 3. 作用域与变量

```python
# 局部变量 vs 全局变量
x = 10  # 全局变量

def func():
    x = 20  # 局部变量
    print(x)  # 20

func()
print(x)  # 10

# 使用 global 修改全局变量
count = 0

def increment():
    global count
    count += 1

increment()
print(count)  # 1

# 使用 nonlocal 修改外层函数的变量
def outer():
    x = 10
    def inner():
        nonlocal x
        x = 20
    inner()
    print(x)  # 20

outer()
```

### 4. Lambda 表达式

```python
# 匿名函数
square = lambda x: x ** 2
print(square(5))  # 25

# 多参数
add = lambda x, y: x + y
print(add(3, 4))  # 7

# 常用场景：作为参数传递
nums = [1, 2, 3, 4, 5]

# 排序
pairs = [(1, 'one'), (2, 'two'), (3, 'three')]
pairs.sort(key=lambda pair: pair[1])

# map
squares = list(map(lambda x: x**2, nums))  # [1, 4, 9, 16, 25]

# filter
evens = list(filter(lambda x: x % 2 == 0, nums))  # [2, 4]

# 条件表达式
get_status = lambda score: "及格" if score >= 60 else "不及格"
```

### 5. 迭代器（Iterator）

迭代器是实现了 `__iter__()` 和 `__next__()` 的对象，可以逐个返回元素。

```python
# 可迭代对象 -> 迭代器
nums = [1, 2, 3]
iterator = iter(nums)

print(next(iterator))  # 1
print(next(iterator))  # 2
print(next(iterator))  # 3
# print(next(iterator))  # StopIteration 异常

# for 循环的本质就是使用迭代器
for num in nums:
    print(num)

# 自定义迭代器
class CountDown:
    def __init__(self, start):
        self.start = start
    
    def __iter__(self):
        return self
    
    def __next__(self):
        if self.start <= 0:
            raise StopIteration
        self.start -= 1
        return self.start + 1

for num in CountDown(5):
    print(num)  # 5 4 3 2 1
```

### 6. 生成器（Generator）

生成器是一种特殊的迭代器，使用 `yield` 关键字，可以暂停和恢复执行，惰性求值，节省内存。

```python
# 生成器函数
def countdown(n):
    while n > 0:
        yield n  # 暂停，返回 n
        n -= 1   # 下次从这里继续

for num in countdown(5):
    print(num)  # 5 4 3 2 1

# 生成器表达式（类似列表推导式，但惰性求值）
squares = (x**2 for x in range(1000000))  # 不占用大量内存
print(next(squares))  # 0
print(next(squares))  # 1

# 对比：列表推导式会立即生成所有元素
squares_list = [x**2 for x in range(1000000)]  # 占用大量内存

# 生成器应用：读取大文件
def read_large_file(file_path):
    with open(file_path, "r", encoding="utf-8") as f:
        for line in f:
            yield line.strip()

# 处理大文件，一行一行读，内存友好
for line in read_large_file("big_file.txt"):
    process(line)

# yield from：委托子生成器
def flatten(nested_list):
    for item in nested_list:
        if isinstance(item, list):
            yield from flatten(item)  # 递归委托
        else:
            yield item

print(list(flatten([1, [2, 3], [4, [5, 6]]])))  # [1, 2, 3, 4, 5, 6]
```

### 7. 装饰器（Decorator）

装饰器是接收函数作为参数并返回函数的高阶函数，用于在不修改原函数的情况下扩展功能。

```python
# 基础装饰器
def my_decorator(func):
    def wrapper(*args, **kwargs):
        print("函数执行前")
        result = func(*args, **kwargs)
        print("函数执行后")
        return result
    return wrapper

@my_decorator
def say_hello():
    print("Hello!")

say_hello()
# 输出：
# 函数执行前
# Hello!
# 函数执行后

# 带参数的装饰器
def repeat(times):
    def decorator(func):
        def wrapper(*args, **kwargs):
            for _ in range(times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(3)
def greet(name):
    print(f"Hello, {name}!")

greet("Alice")  # 打印 3 次

# 常用装饰器：计时
import time

def timer(func):
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        print(f"{func.__name__} 耗时: {elapsed:.4f}秒")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1)
    return "Done"

slow_function()

# 常用装饰器：缓存（LRU 缓存）
from functools import lru_cache

@lru_cache(maxsize=128)
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

print(fibonacci(100))  # 极快，因为缓存了中间结果

# 保留原函数信息
from functools import wraps

def my_decorator(func):
    @wraps(func)  # 保留原函数的 __name__、__doc__ 等
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper
```

### 8. 函数速查表

| 特性 | 语法 | 用途 |
|------|------|------|
| 默认参数 | `def f(a=1)` | 提供默认值 |
| 可变参数 | `*args` | 接收任意数量位置参数 |
| 关键字参数 | `**kwargs` | 接收任意数量关键字参数 |
| Lambda | `lambda x: x+1` | 简单匿名函数 |
| 迭代器 | `__iter__`, `__next__` | 自定义遍历逻辑 |
| 生成器 | `yield` | 惰性求值，节省内存 |
| 装饰器 | `@decorator` | 扩展函数功能 |

```python
class MyIterable:
    def __init__(self, data):
        self.data = data
    def __iter__(self):
        return MyIterator(self.data)

class MyIterator:
    def __init__(self, data):
        self.data = list(data)
        self.index = 0
    def __iter__(self):
        return self
    def __next__(self):
        if self.index < len(self.data):
            result = self.data[self.index]
            self.index += 1
            return result
        raise StopIteration

container1 = MyIterator([1, 2, 3])
print(next(container1))
```

```python
# 求余数
def func(a, b):

# 三角形面积
import math

def s(a, b, c):
    return (a + b + c) / 2

def area(a, b, c):
    S = s(a, b, c)
    result = math.sqrt(S * (S - a) * (S - b) * (S - c))
    print(f"{result:.3f}")

# 三数最大值
def count(nums):
    result = nums[0]
    for i in nums:
        if i > result:
            result = i
    print(f"{result:.3f}")

# 矩阵转置
def row_col(arr):
    n = len(arr)
    return [[arr[j][i] for j in range(n)] for i in range(n)]

# 字符串统计
def func_str(a: str):
    p = n = s = o = 0
    for i in a:
        if i.isdigit():
            n += 1
        elif i.isalpha():
            p += 1
        elif i.isspace():
            s += 1
        else:
            o += 1
    return p, n, s, o

# 递归：母牛数量
def count_cow(n):
    if 0 < n < 55:
        if n == 1:
            return 1
        elif n == 2:
            return 2
        elif n == 3:
            return 3
        return count_cow(n - 1) + count_cow(n - 3)
    else:
        raise ValueError('输入不合法')

# 数组追加
def add_arr(a, m1, b, n1):
    a[m1:] = b[:n1]
    return a

# 最小最大交换
def func_input():
    return list(map(int, input().strip().split()))

def func_process(lst: list):
    result = lst.copy()
    min_idx = result.index(min(result))
    max_idx = result.index(max(result))
    result[min_idx], result[0] = result[0], result[min_idx]
    if max_idx == 0:
        max_idx = min_idx
    result[max_idx], result[-1] = result[-1], result[max_idx]
    return result

# 矩阵中绝对值最大元素
def max_count(arr):
    max_val = arr[0][0]
    max_row = 1
    max_col = 1
    for row in range(len(arr)):
        cur_row_max = max(arr[row], key=abs)
        if abs(cur_row_max) > abs(max_val):
            max_val = cur_row_max
            max_row = row + 1
            max_col = arr[row].index(max_val) + 1
    return max_val, max_row, max_col

# 亲密数
def factor_sum(n):
    return sum(i for i in range(1, n) if n % i == 0)

factor_dict = {i: factor_sum(i) for i in range(1, 3001)}
result = []
for a in range(1, 3001):
    b = factor_dict.get(a)
    if b > a and b <= 3000 and factor_dict.get(b) == a:
        result.append(f"({a},{b})")
print(''.join(result))
```
