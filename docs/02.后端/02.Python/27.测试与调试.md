# 测试与调试

## 1. 单元测试 unittest

```python
import unittest

def add(a, b):
    return a + b

def divide(a, b):
    if b == 0:
        raise ValueError("除数不能为零")
    return a / b

class TestMathOperations(unittest.TestCase):
    def setUp(self):
        self.numbers = [1, 2, 3, 4, 5]
    
    def test_add_positive(self):
        self.assertEqual(add(2, 3), 5)
    
    def test_divide(self):
        self.assertAlmostEqual(divide(10, 3), 3.333, places=3)
    
    def test_divide_by_zero(self):
        with self.assertRaises(ValueError):
            divide(10, 0)
    
    def test_in_list(self):
        self.assertIn(3, self.numbers)
        self.assertNotIn(10, self.numbers)

if __name__ == "__main__":
    unittest.main()
```

## 2. pytest（更简洁的测试框架）

```python
import pytest

def add(a, b):
    return a + b

def test_add():
    assert add(2, 3) == 5

# 参数化测试
@pytest.mark.parametrize("a,b,expected", [
    (2, 3, 5),
    (-1, 1, 0),
    (0, 0, 0),
    (100, 200, 300)
])
def test_add_parametrized(a, b, expected):
    assert add(a, b) == expected

# 使用 fixture
@pytest.fixture
def sample_data():
    return {"name": "Alice", "age": 25}

def test_data(sample_data):
    assert sample_data["name"] == "Alice"
```

## 3. 调试技巧

```python
import pdb

def buggy_function():
    x = 10
    y = 0
    pdb.set_trace()  # 设置断点
    result = x / y
    return result

# 常用 pdb 命令：
# n(next) - 执行下一行
# s(step) - 进入函数
# c(continue) - 继续执行
# p variable - 打印变量
# l(list) - 显示代码
# q(quit) - 退出调试

# 使用 breakpoint()（Python 3.7+）
def modern_debug():
    x = [1, 2, 3]
    breakpoint()
    print(sum(x))

# 日志调试
import logging
logging.basicConfig(level=logging.DEBUG)
logger = logging.getLogger(__name__)

def logged_function():
    logger.debug("进入函数")
    result = 42
    logger.info(f"计算结果: {result}")
    return result
```
