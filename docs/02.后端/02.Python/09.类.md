### 1. 定义类

```python
class Dog:
    """狗的类"""
    
    # 类属性（所有实例共享）
    species = "Canis familiaris"
    
    # 构造方法：创建实例时自动调用
    def __init__(self, name, age):
        self.name = name      # 实例属性
        self.age = age
    
    # 实例方法
    def bark(self):
        return f"{self.name} says woof!"
    
    def get_info(self):
        return f"{self.name} is {self.age} years old"

# 创建实例
my_dog = Dog("Buddy", 3)
print(my_dog.name)       # Buddy
print(my_dog.bark())     # Buddy says woof!
print(my_dog.species)    # Canis familiaris
```

### 2. 特殊方法（魔术方法）

| 方法 | 触发时机 | 示例 |
|------|----------|------|
| `__init__` | 创建实例时 | `obj = Class()` |
| `__str__` | `str()` 或 `print()` | 返回可读字符串 |
| `__repr__` | `repr()` 或交互式解释器 | 返回开发字符串 |
| `__eq__` | `==` 比较 | 自定义相等逻辑 |
| `__lt__` | `<` 比较 | 自定义排序 |
| `__len__` | `len()` | 返回长度 |
| `__getitem__` | `obj[key]` | 索引访问 |
| `__call__` | `obj()` | 使实例可调用 |
| `__add__` | `+` 运算符 | 自定义加法 |
| `__iter__` | `for` 循环 | 使实例可迭代 |

```python
class Book:
    def __init__(self, title, author, pages):
        self.title = title
        self.author = author
        self.pages = pages
    
    def __str__(self):
        return f"《{self.title}》 by {self.author}"
    
    def __repr__(self):
        return f"Book('{self.title}', '{self.author}', {self.pages})"
    
    def __eq__(self, other):
        return self.title == other.title and self.author == other.author
    
    def __len__(self):
        return self.pages
    
    def __add__(self, other):
        return Book(f"{self.title} & {other.title}", "合集", self.pages + other.pages)

book = Book("Python", "Alice", 300)
print(book)           # 《Python》 by Alice
print(repr(book))     # Book('Python', 'Alice', 300)
print(len(book))      # 300
```

### 3. 类属性 vs 实例属性

```python
class Student:
    school = "ABC School"  # 类属性（共享）
    count = 0              # 类属性（计数器）
    
    def __init__(self, name):
        self.name = name   # 实例属性（独立）
        Student.count += 1 # 修改类属性

s1 = Student("Alice")
s2 = Student("Bob")

print(s1.school)   # ABC School
print(s2.school)   # ABC School
print(Student.count)  # 2

# 通过实例修改类属性？实际上是创建实例属性！
s1.school = "XYZ"  # 创建实例属性 s1.school
print(s1.school)   # XYZ（实例属性）
print(s2.school)   # ABC School（类属性未变）
print(Student.school)  # ABC School
```

### 4. 类方法、静态方法

```python
class DateUtil:
    def __init__(self, year, month, day):
        self.year = year
        self.month = month
        self.day = day
    
    # 实例方法：操作实例
    def format(self):
        return f"{self.year}-{self.month:02d}-{self.day:02d}"
    
    # 类方法：操作类，第一个参数是 cls
    @classmethod
    def from_string(cls, date_str):
        year, month, day = map(int, date_str.split("-"))
        return cls(year, month, day)
    
    # 静态方法：不依赖类或实例
    @staticmethod
    def is_valid_date(year, month, day):
        if month < 1 or month > 12:
            return False
        if day < 1 or day > 31:
            return False
        return True

# 使用
d = DateUtil.from_string("2024-3-15")
print(d.format())  # 2024-03-15

print(DateUtil.is_valid_date(2024, 13, 1))  # False
```

### 5. 属性装饰器

```python
class Circle:
    def __init__(self, radius):
        self._radius = radius  # 约定：单下划线表示内部使用
    
    @property
    def radius(self):
        """获取半径"""
        return self._radius
    
    @radius.setter
    def radius(self, value):
        """设置半径（带验证）"""
        if value < 0:
            raise ValueError("半径不能为负数")
        self._radius = value
    
    @property
    def area(self):
        """计算面积（只读属性）"""
        import math
        return math.pi * self._radius ** 2

# 使用
c = Circle(5)
print(c.radius)   # 5（像访问属性一样调用方法）
c.radius = 10     # 调用 setter
print(c.area)     # 314.159...（只读）
# c.area = 100    # 错误，没有 setter
```

### 6. 私有属性与名称改写

```python
class BankAccount:
    def __init__(self, owner, balance):
        self.owner = owner
        self.__balance = balance  # 双下划线：名称改写
    
    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount
            return True
        return False
    
    def get_balance(self):
        return self.__balance

account = BankAccount("Alice", 1000)
# print(account.__balance)      # AttributeError
print(account.get_balance())    # 1000
print(account._BankAccount__balance)  # 1000（可以访问，但不建议）
```
