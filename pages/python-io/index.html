    Python输入与输出

    # 输入与输出

    # 1. 输出：print()

    print() 函数用于将内容输出到屏幕。

    # 基础输出
print("Hello, World!")

# 输出多个值，用逗号分隔，默认以空格连接
print("姓名:", "Alice", "年龄:", 20)  # 姓名: Alice 年龄: 20

# 自定义分隔符
print("a", "b", "c", sep="-")  # a-b-c

# 自定义结束符（默认换行）
print("第一行", end=" ")
print("第二行")  # 第一行 第二行

# 格式化输出 - f-string（推荐）
name = "Alice"
age = 20
print(f"我叫{name}，今年{age}岁")

# 格式化输出 - format()
print("我叫{}，今年{}岁".format(name, age))

# 格式化输出 - % 运算符（旧式）
print("我叫%s，今年%d岁" % (name, age))

    # 2. 输入：input()

    input() 函数用于接收用户的键盘输入，返回值始终是字符串 str 类型。

    # 基础输入
name = input("请输入你的名字: ")
print(f"你好, {name}!")

# 输入数字需要转换类型
age = int(input("请输入你的年龄: "))
height = float(input("请输入你的身高: "))

# 一次输入多个值
a, b = input("输入两个数，用空格分隔: ").split()
a = int(a)
b = int(b)

# 输入列表
nums = list(map(int, input("输入多个整数，用空格分隔: ").split()))

age = 33
weight = 74.45

    "姓名 : {0}\n"
    "年龄 : {1}"

print(talk)
