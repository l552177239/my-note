# 默认参数不要用可变对象！
def bad_append(item, lst=[]):  # 错误！
    lst.append(item)
    return lst

def good_append(item, lst=None):  # 正确
    if lst is None:
        lst = []
    lst.append(item)
    return lst

    return sum(args)

# 可变关键字参数 **kwargs
def print_info(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

    # 5. 迭代器（Iterator）

    迭代器是实现了 __iter__() 和 __next__() 的对象，可以逐个返回元素。

    # 可迭代对象 -> 迭代器
nums = [1, 2, 3]
iterator = iter(nums)

print(next(iterator))  # 1

# print(next(iterator))  # StopIteration 异常

# for 循环的本质就是使用迭代器
for num in nums:
    print(num)

# 自定义迭代器
class CountDown:
    def __init__(self, start):
        self.start = start
    
    def __iter__(self):
        return self
    
    def __next__(self):
        if self.start <= 0:
            raise StopIteration
        self.start -= 1
        return self.start + 1

for num in CountDown(5):
    print(num)  # 5 4 3 2 1

    # 6. 生成器（Generator）

    生成器是一种特殊的迭代器，使用 yield 关键字，可以暂停和恢复执行，惰性求值，节省内存。

    # 生成器函数
def countdown(n):
    while n > 0:
        yield n  # 暂停，返回 n
        n -= 1   # 下次从这里继续

for num in countdown(5):
    print(num)  # 5 4 3 2 1

# 生成器表达式（类似列表推导式，但惰性求值）
squares = (x**2 for x in range(1000000))  # 不占用大量内存
print(next(squares))  # 0
print(next(squares))  # 1

# 对比：列表推导式会立即生成所有元素
squares_list = [x**2 for x in range(1000000)]  # 占用大量内存

# 生成器应用：读取大文件
def read_large_file(file_path):
    with open(file_path, "r", encoding="utf-8") as f:
        for line in f:
            yield line.strip()

# 处理大文件，一行一行读，内存友好
for line in read_large_file("big_file.txt"):
    process(line)

# yield from：委托子生成器
def flatten(nested_list):
    for item in nested_list:
        if isinstance(item, list):
            yield from flatten(item)  # 递归委托
        else:
            yield item

print(list(flatten([1, [2, 3], [4, [5, 6]]])))  # [1, 2, 3, 4, 5, 6]

    # 7. 装饰器（Decorator）

    装饰器是接收函数作为参数并返回函数的高阶函数，用于在不修改原函数的情况下扩展功能。

    # 基础装饰器
def my_decorator(func):
    def wrapper(*args, **kwargs):
        print("函数执行前")
        result = func(*args, **kwargs)
        print("函数执行后")
        return result
    return wrapper

@my_decorator
def say_hello():
    print("Hello!")

say_hello()
# 输出：
# 函数执行前
# Hello!
# 函数执行后

# 带参数的装饰器
def repeat(times):
    def decorator(func):
        def wrapper(*args, **kwargs):
            for _ in range(times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(3)
def greet(name):
    print(f"Hello, {name}!")

greet("Alice")  # 打印 3 次

# 常用装饰器：计时
import time

def timer(func):
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        print(f"{func.__name__} 耗时: {elapsed:.4f}秒")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1)
    return "Done"

slow_function()

# 常用装饰器：缓存（LRU 缓存）
from functools import lru_cache

@lru_cache(maxsize=128)
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

print(fibonacci(100))  # 极快，因为缓存了中间结果

# 保留原函数信息
from functools import wraps

def my_decorator(func):
    @wraps(func)  # 保留原函数的 __name__、__doc__ 等
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

    # 迭代器

    class MyIterable:
    def __init__(self, data):
        self.data = data
    def __iter__(self):
        return MyIterator(self.data)

class MyIterator:
    def __init__(self, data):
        self.data = list(data)
        self.index = 0
    def __iter__(self):
        return self
    def __next__(self):
        if self.index < len(self.data):
            result = self.data[self.index]
            self.index += 1
            return result
        raise StopIteration

container1 = MyIterator([1, 2, 3])
print(next(container1))

# 三角形面积
import math

def s(a, b, c):
    return (a + b + c) / 2

def area(a, b, c):
    S = s(a, b, c)
    result = math.sqrt(S * (S - a) * (S - b) * (S - c))
    print(f"{result:.3f}")

# 三数最大值
def count(nums):
    result = nums[0]
    for i in nums:
        if i > result:
            result = i
    print(f"{result:.3f}")

# 矩阵转置
def row_col(arr):
    n = len(arr)
    return [[arr[j][i] for j in range(n)] for i in range(n)]

# 字符串统计
def func_str(a: str):
    p = n = s = o = 0
    for i in a:
        if i.isdigit():
            n += 1
        elif i.isalpha():
            p += 1
        elif i.isspace():
            s += 1
        else:
            o += 1
    return p, n, s, o

# 递归：母牛数量
def count_cow(n):
    if 0 < n < 55:
        if n == 1:
            return 1
        elif n == 2:
            return 2
        elif n == 3:
            return 3
        return count_cow(n - 1) + count_cow(n - 3)
    else:
        raise ValueError('输入不合法')

# 最小最大交换
def func_input():
    return list(map(int, input().strip().split()))

def func_process(lst: list):
    result = lst.copy()
    min_idx = result.index(min(result))
    max_idx = result.index(max(result))
    result[min_idx], result[0] = result[0], result[min_idx]
    if max_idx == 0:
        max_idx = min_idx
    result[max_idx], result[-1] = result[-1], result[max_idx]
    return result

# 矩阵中绝对值最大元素
def max_count(arr):
    max_val = arr[0][0]
    max_row = 1
    max_col = 1
    for row in range(len(arr)):
        cur_row_max = max(arr[row], key=abs)
        if abs(cur_row_max) > abs(max_val):
            max_val = cur_row_max
            max_row = row + 1
            max_col = arr[row].index(max_val) + 1
    return max_val, max_row, max_col

# 亲密数
def factor_sum(n):
    return sum(i for i in range(1, n) if n % i == 0)

factor_dict = {i: factor_sum(i) for i in range(1, 3001)}
result = []
for a in range(1, 3001):
    b = factor_dict.get(a)
    if b > a and b <= 3000 and factor_dict.get(b) == a:
        result.append(f"({a},{b})")
print(''.join(result))
