    # 1. 基础语法：if - elif - else

    age = 18

if age < 13:
    print("儿童")
elif age < 20:
    print("青少年")  # 输出：青少年
else:
    print("成年人")

    # 2. 比较运算符

    运算符 含义 示例 结果
    == 等于 5 == 5 True
    != 不等于 5 != 3 True
    > 大于 5 > 3 True
    < 小于 5 < 3 False
    >= 大于等于 5 >= 5 True
    <= 小于等于 5 <= 3 False

    # 3. 逻辑运算符

    运算符 含义 示例 说明
    and 与 True and False 两边都为 True 才为 True
    or 或 True or False 只要一边为 True 就为 True
    not 非 not True 取反
    age = 25
income = 5000

# and: 两边都要满足
if age >= 18 and income >= 3000:
    print("符合申请条件")

# or: 满足其一即可
if age < 18 or age > 60:
    print("享受优惠票价")

# not: 取反
is_raining = False
if not is_raining:
    print("今天不下雨，可以出门")

    # 5. 身份运算符

    运算符 含义 示例 说明
    is 是同一个对象 a is b 判断两个变量是否引用同一对象
    is not 不是同一个对象 a is not b 
    a = [1, 2, 3]
b = a
c = [1, 2, 3]

print(a is b)      # True（同一对象）
print(a is c)      # False（值相同，但不是同一对象）
print(a == c)      # True（值相等）

    # 6. 条件表达式（三元运算符）

    # 语法：值1 if 条件 else 值2
age = 20
status = "成年" if age >= 18 else "未成年"
print(status)  # 成年

# 等价于
if age >= 18:
    status = "成年"
else:
    status = "未成年"

    # 7. 真值判断

    在 Python 中，以下值在条件判断中被视为 False：

    • None
    • False
    • 0（整数零）
    • 0.0（浮点零）
    • ""（空字符串）
    • []（空列表）
    • {}（空字典）
    • ()（空元组）
    • set()（空集合）

    其他所有值都被视为 True。

    name = ""
if name:
    print("名字不为空")
else:
    print("名字为空")  # 输出

nums = []
if nums:
    print("列表不为空")
else:
    print("列表为空")  # 输出

# 分段函数
from math import fabs, pow

def f(x):
    if x < 0:
        result = fabs(x)
    elif 0 <= x < 2:
        result = pow(x + 1, 0.5)
    elif 2 <= x < 4:
        result = pow(x + 2, 5)
    else:
        result = 2 * x + 5
    print(f"{result:.2f}")

# 剪刀石头布

def LEAP_YEAR(y):

# 购物车结算
carts = [
    {"name": "床", "price": 1999, "num": 1},
    {"name": "枕头", "price": 10, "num": 2},
    {"name": "被子", "price": 20, "num": 1},
]
money = int(input().strip())
shop_money = sum(x["num"] * x["price"] for x in carts)
if shop_money > money:
    print("账户余额不足")
else:
    print("购买成功")
