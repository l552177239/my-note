# 魔术方法大全

方法 触发时机 用途
__new__ 创建实例之前 控制实例创建（元类）
__init__ 创建实例之后 初始化实例
__del__ 实例被销毁 析构（不建议依赖）
__repr__ repr(obj) 开发表示
__str__ str(obj) / print() 用户可读表示
__bytes__ bytes(obj) 字节表示
__format__ format(obj) 格式化
__hash__ hash(obj) 哈希值（用于 dict/set）
__bool__ bool(obj) 真值判断
__eq__ == 等于
__ne__ != 不等于
__lt__ < 小于
__le__ <= 小于等于
__gt__ > 大于
__ge__ >= 大于等于
__add__ + 加法
__sub__ - 减法
__mul__ * 乘法
__truediv__ / 真除法
__floordiv__ // 地板除
__mod__ % 取模
__pow__ ** 幂运算
__len__ len(obj) 长度
__getitem__ obj[key] 获取项
__setitem__ obj[key] = val 设置项
__delitem__ del obj[key] 删除项
__iter__ for x in obj 迭代
__next__ next(iterator) 下一个元素
__contains__ in 成员判断
__call__ obj() 可调用
__enter__ with obj 上下文进入
__exit__ 离开 with 上下文退出
__getattr__ 访问不存在的属性 动态属性

__delattr__ 删除属性 属性删除拦截
__slots__ 类定义时 限制属性，节省内存

# 上下文管理器 with 语句

# 自定义上下文管理器
class DatabaseConnection:
    def __init__(self, connection_string):
        self.connection_string = connection_string
        self.connection = None
    
    def __enter__(self):
        print(f"连接到: {self.connection_string}")
        self.connection = "连接对象"
        return self.connection
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        print("关闭连接")
        if exc_type:
            print(f"发生异常: {exc_val}")
        return True  # 抑制异常

with DatabaseConnection("mysql://localhost") as conn:
    print(f"使用连接: {conn}")

# 使用 contextmanager 装饰器
from contextlib import contextmanager

@contextmanager
def managed_resource(name):
    print(f"获取资源: {name}")
    resource = f"资源-{name}"
    try:
        yield resource
    finally:
        print(f"释放资源: {name}")

with managed_resource("数据库") as res:
    print(f"使用: {res}")

# 元类（Metaclass）

# 元类是创建类的类
class SingletonMeta(type):
    """单例元类"""
    _instances = {}
    
    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

class Database(metaclass=SingletonMeta):
    def __init__(self):
        print("初始化数据库")

db1 = Database()
db2 = Database()
print(db1 is db2)  # True
