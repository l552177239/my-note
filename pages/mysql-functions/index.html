# 常用函数

整理字符串、数值、日期与条件类常用函数。

# 1. 字符串函数

SELECT CONCAT('Hello', ' ', 'World');     -- Hello World
SELECT LENGTH('Hello');                      -- 5
SELECT SUBSTRING('Hello World', 1, 5);       -- Hello
SELECT REPLACE('Hello World', 'World', 'SQL'); -- Hello SQL
SELECT UPPER('hello');                       -- HELLO
SELECT LOWER('HELLO');                       -- hello
SELECT TRIM('  hello  ');                    -- hello
SELECT LEFT('Hello', 3);                     -- Hel
SELECT RIGHT('Hello', 3);                    -- llo
SELECT INSTR('Hello World', 'World');        -- 7

# 3. 日期函数

SELECT NOW();                          -- 当前日期时间
SELECT CURDATE();                      -- 当前日期
SELECT CURTIME();                      -- 当前时间
SELECT YEAR('2024-01-15');             -- 2024
SELECT MONTH(NOW());                   -- 当前月份
SELECT DAY(NOW());                     -- 当前日
SELECT DATE_ADD(NOW(), INTERVAL 7 DAY); -- 7天后
SELECT DATEDIFF('2024-12-31', '2024-01-01'); -- 365
SELECT DATE_FORMAT(NOW(), '%Y-%m-%d %H:%i:%s'); -- 格式化日期

# 4. 条件函数

-- IF
SELECT IF(salary > 10000, '高薪', '普通') FROM employees;

-- IFNULL
SELECT IFNULL(email, '未填写') FROM employees;

-- CASE WHEN
SELECT 
    name,
    salary,
    CASE 
        WHEN salary >= 20000 THEN '高级'
        WHEN salary >= 10000 THEN '中级'
        ELSE '初级'
    END AS 级别
FROM employees;
